# Python deps for the Chiemsee DWD ICON weather pipeline.
# Pinned for reproducible GitHub Actions runs on ubuntu-latest.
#
# GRIB2 is decoded with cfgrib -> ecCodes. We install the system ecCodes
# library in the workflow (libeccodes-dev) AND pin the eccodes Python binding;
# recent eccodes wheels also ship the binary lib via eccodeslib, so either path
# works. cfgrib maps GRIB messages onto xarray datasets.

# requests is used by BOTH build_weather.py (GRIB downloads) and
# build_stormlight.py (WarnWetter feed). gzip handling in build_stormlight.py is
# stdlib -- no extra dependency needed for the storm-warning-light JSON.
requests==2.32.3
numpy==2.1.3
xarray==2024.11.0
cfgrib==0.9.14.1
eccodes==2.38.3
# tzdata provides the Europe/Berlin zone for zoneinfo on minimal runners.
tzdata==2024.2

# ⚠️ paramiko lädt die fertigen JSON-Dateien per SFTP auf den Webspace
# (upload_sftp.py). Im alten Ablauf wurde es in einem EIGENEN Schritt per
# "pip install paramiko" nachinstalliert und stand deshalb nie hier drin. Beim
# Zusammenlegen zu einem Ablauf ging diese Zeile verloren — alle fünf Seen
# wurden gebaut und keiner hochgeladen (2026-07-28). Hier gehört es hin.
#
# Bewusst 3.x und nicht die neueste 5.0: Das ist die Reihe, mit der die alte
# Pipeline monatelang lief. Ein Sprung über zwei Hauptversionen gehört nicht in
# einen Fehlerbehebung.
paramiko==3.5.1
